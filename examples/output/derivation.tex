\section*{every dog who chased Felix barks}

\bigskip
\begin{align*}
\llbracket \text{\textit{chased} \textit{Felix}} \rrbracket &= (\lambda y.\lambda x.\mathit{chase}(x, y))(\mathit{felix}) && \text{(FA)} \\
&= \lambda x.\mathit{chase}(x, \mathit{felix}) && (\beta)
\end{align*}
\begin{align*}
\llbracket \text{\textit{t$_{1}$} \textit{chased} \textit{Felix}} \rrbracket &= (\lambda x.\mathit{chase}(x, \mathit{felix}))(x_{1}) && \text{(FA)} \\
&= \mathit{chase}(x_{1}, \mathit{felix}) && (\beta)
\end{align*}
\begin{align*}
\llbracket \text{$\lambda_{1}$ \textit{t$_{1}$} \textit{chased} \textit{Felix}} \rrbracket &= \lambda x_{1}.\mathit{chase}(x_{1}, \mathit{felix}) && \text{(PA)}
\end{align*}
\begin{align*}
\llbracket \text{\textit{who} $\lambda_{1}$ \textit{t$_{1}$} \textit{chased} \textit{Felix}} \rrbracket &= (\lambda P.P)(\lambda x_{1}.\mathit{chase}(x_{1}, \mathit{felix})) && \text{(FA)} \\
&= \lambda x_{1}.\mathit{chase}(x_{1}, \mathit{felix}) && (\beta)
\end{align*}
\begin{align*}
\llbracket \text{\textit{dog} \textit{who} $\lambda_{1}$ \textit{t$_{1}$} \textit{chased} \textit{Felix}} \rrbracket &= \lambda x.(\lambda x.\mathit{dog}(x))(x) \wedge (\lambda x_{1}.\mathit{chase}(x_{1}, \mathit{felix}))(x) && \text{(PM)} \\
&= \lambda x.\mathit{dog}(x) \wedge (\lambda x_{1}.\mathit{chase}(x_{1}, \mathit{felix}))(x) && (\beta) \\
&= \lambda x.\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix}) && (\beta)
\end{align*}
\begin{align*}
\llbracket \text{\textit{every} \textit{dog} \textit{who} $\lambda_{1}$ \textit{t$_{1}$} \textit{chased} \textit{Felix}} \rrbracket &= (\lambda P.\lambda Q.\forall x[P(x) \rightarrow Q(x)])(\lambda x.\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix})) && \text{(FA)} \\
&= \lambda Q.\forall x[(\lambda x.\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix}))(x) \rightarrow Q(x)] && (\beta) \\
&= \lambda Q.\forall x[\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix}) \rightarrow Q(x)] && (\beta)
\end{align*}
\begin{align*}
\llbracket \text{\textit{every} \textit{dog} \textit{who} $\lambda_{1}$ \textit{t$_{1}$} \textit{chased} \textit{Felix} \textit{barks}} \rrbracket &= (\lambda Q.\forall x[\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix}) \rightarrow Q(x)])(\lambda x.\mathit{bark}(x)) && \text{(FA)} \\
&= \forall x[\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix}) \rightarrow (\lambda x.\mathit{bark}(x))(x)] && (\beta) \\
&= \forall x[\mathit{dog}(x) \wedge \mathit{chase}(x, \mathit{felix}) \rightarrow \mathit{bark}(x)] && (\beta)
\end{align*}
